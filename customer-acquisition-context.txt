type IncomingMessage = {
  role: "assistant" | "user";
  content: string;
};

type AssessmentOutput = {
  message: string;
  assessment: {
    assessmentComplete?: boolean;
    businessName?: string | null;
    industry?: string | null;
    businessModel?: string | null;
    teamSize?: string | null;
    decisionMaker?: string | null;
    businessProfile?: Record<string, unknown>;
    operationalChallenges?: unknown[];
    currentTools?: unknown[];

    assessmentSummary?: string;
  };
};

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function cleanString(value: unknown): string | null {
  if (typeof value !== "string") return null;
  const result = value.trim();
  return result.length ? result.slice(0, 2000) : null;
}

function cleanArray(value: unknown): Prisma.InputJsonArray {
  if (!Array.isArray(value)) return [];
  const parsed: unknown = JSON.parse(JSON.stringify(value.slice(0, 100)));
  return Array.isArray(parsed) ? (parsed as Prisma.InputJsonArray) : [];
}

function cleanObject(value: unknown): Prisma.InputJsonObject {
  if (!isRecord(value)) return {};
  const parsed: unknown = JSON.parse(JSON.stringify(value));
  return isRecord(parsed) ? (parsed as Prisma.InputJsonObject) : {};
}

function validEmail(value: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value) && value.length <= 254;
}

function getLatestUserMessage(messages: IncomingMessage[]): string {
  return messages[messages.length - 1].content.trim();
}

function isPositiveConsent(message: string): boolean {
  const value = message.toLowerCase().trim();

  return /^(yes|yes please|yeah|yep|sure|okay|ok|i agree|i would|i'd like to|let's book|lets book|book a consultation|i want to book|i'm interested|im interested|proceed|continue)$/i.test(
    value
  );
}

function isNegativeConsent(message: string): boolean {
  const value = message.toLowerCase().trim();

  return /^(no|no thanks|not now|not yet|maybe later|i decline|don't book|do not book|not interested|stop|cancel)$/i.test(
    value
  );
}

function htmlParagraphs(value: string): string {
  return value
    .split(/\n+/)
    .map((line) => `<p>${escapeHtml(line)}</p>`)
    .join("");
}

async function notifyAdmin(
  subject: string,
  text: string,
  html: string
): Promise<void> {
  const to = getAdminNotificationEmail();

  if (!to) {
    console.warn("6ixtem admin notification skipped: no notification email configured.");
    return;
  }

  try {
    await send6ixtemEmail({ to, subject, text, html });
  } catch (error) {
    // Notification failure must not undo a saved assessment or lead.
    console.error("6ixtem admin notification failed:", error);
  }
}

export async function POST(request: Request) {
  let stage = "request validation";

  try {
    if (!process.env.OPENAI_API_KEY) {
      return NextResponse.json(
        { error: "The AI assessment service is not configured." },
        { status: 503 }
      );
    }

    if (!process.env.DATABASE_URL) {
      return NextResponse.json(
        { error: "The assessment database is not configured." },
        { status: 503 }
      );
    }

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        { error: "The request body must contain valid JSON." },
        { status: 400 }
      );
    }

    if (!isRecord(body)) {
      return NextResponse.json(
        { error: "Invalid request body." },
        { status: 400 }
      );
    }

    const sessionId =
      typeof body.sessionId === "string" ? body.sessionId : "";

    const investmentReadiness =
      typeof body.investmentReadiness === "string"
        ? body.investmentReadiness
        : "maybe";

    const incomingMessages = body.messages;

    if (!/^[a-zA-Z0-9_-]{10,100}$/.test(sessionId)) {
      return NextResponse.json(
        { error: "Invalid assessment session." },
        { status: 400 }
      );
    }

    if (!["yes", "maybe", "no"].includes(investmentReadiness)) {
      return NextResponse.json(
        { error: "Invalid investment readiness value." },
        { status: 400 }
      );
    }

    if (!Array.isArray(incomingMessages) || incomingMessages.length === 0) {
      return NextResponse.json(
        { error: "A conversation message is required." },
        { status: 400 }
      );
    }

    if (incomingMessages.length > 80) {
      return NextResponse.json(
        { error: "The assessment has reached its message limit." },
        { status: 400 }
      );
    }

    const messages: IncomingMessage[] = [];

    for (const item of incomingMessages) {
      if (
        !isRecord(item) ||
        !["assistant", "user"].includes(String(item.role)) ||
        typeof item.content !== "string" ||
        !item.content.trim() ||
        item.content.length > 5000
      ) {
        return NextResponse.json(
          { error: "One or more messages are invalid." },
          { status: 400 }
        );
      }

      messages.push({
        role: item.role as IncomingMessage["role"],
        content: item.content.trim(),
      });
    }

    if (messages[messages.length - 1].role !== "user") {
      return NextResponse.json(
        { error: "The latest message must come from the prospect." },
        { status: 400 }
      );
    }

    const latestMessage = getLatestUserMessage(messages);

    stage = "database lookup";

    const existing = await prisma.assessment.findUnique({
      where: { sessionId },
      select: {
        id: true,
        status: true,
        qualificationStatus: true,
        qualificationScore: true,
        completedAt: true,
        assessmentSummary: true,
        businessName: true,
        industry: true,
        businessModel: true,
        teamSize: true,
        decisionMaker: true,
        investmentReadiness: true,
        businessProfile: true,
        operationalChallenges: true,
        currentTools: true,
        requirements: true,
        contactName: true,
        contactEmail: true,
        funnelState: true,
        assessmentEmailSentAt: true,
        bookingLinkSentAt: true,
      },
    });

    /*
     * Post-assessment funnel:
     * qualification and state are read from the database, never trusted
     * from browser-supplied fields.
     */
    if (existing?.status === "COMPLETED") {
      const canBook = existing.qualificationStatus === "QUALIFIED";

      if (!canBook) {
        return NextResponse.json({
          message:
            "Your assessment is complete. Thank you for sharing information about your business. Based on the information provided, a consultation is not the recommended next step at this time.",
          qualification: {
            status: existing.qualificationStatus,
            score: existing.qualificationScore,
            reasons: [],
          },
          assessment: {
            id: existing.id,
            status: existing.status,
            completed: true,
          },
          nextStep: "ASSESSMENT_CLOSED",
          bookingUrl: null,
          persisted: true,
        });
      }

      if (existing.funnelState === "DECLINED" ||
          existing.funnelState === "CLOSED") {
        return NextResponse.json({
          message: "Thank you for your time. You can return whenever you are ready to explore your business requirements further.",
          qualification: {
            status: existing.qualificationStatus,
            score: existing.qualificationScore,
            reasons: [],
          },
          assessment: {
            id: existing.id,
            status: existing.status,

        status: true,

    });

    const canBook =
      saved.status === "COMPLETED" &&
      saved.qualificationStatus === "QUALIFIED";

    /*
     * Notify the admin when an assessment first completes.
     * Use the persisted timestamp to avoid routine duplicate notifications.

    if (completed && !existing?.completedAt) {

          ].join("\n"),
          emailLayout(`
            <h1>Assessment completed</h1>

            ${htmlParagraphs(normalized.assessmentSummary || "Not provided")}
          `)
        );

        // A failed email is logged; it does not invalidate the saved assessment.
        await prisma.assessment.update({
          where: { id: saved.id },
          data: { assessmentEmailSentAt: sentAt },
        });
      }
    }

    const nextStep = !completed
      ? "CONTINUE_ASSESSMENT"
      : canBook
        ? "AWAITING_CONSENT"
        : "ASSESSMENT_CLOSED";

    return NextResponse.json({
      message: output.message.trim(),
      qualification: {
        status: saved.qualificationStatus,
        score: saved.qualificationScore,
        reasons: qualification.reasons,
      },
      assessment: {
        id: saved.id,
        status: saved.status,
        completed: saved.status === "COMPLETED",
      },
      nextStep,
      bookingUrl: null,
      persisted: true,
    });
  } catch (error) {
    console.error("6ixtem assessment failure:", {
      stage,
      message: error instanceof Error ? error.message : String(error),
      stack: error instanceof Error ? error.stack : undefined,
      name: error instanceof Error ? error.name : undefined,
    });

    return NextResponse.json(
      { error: "We could not process that response. Please try again." },
      { status: 500 }
    );
  }
}

function isRecord(
  value: unknown,
): value is Record<string, unknown> {
  return (
    typeof value === "object" &&
    value !== null &&
    !Array.isArray(value)
  );
}

function isValidEmail(value: string): boolean {
  return (
    value.length <= 254 &&
    /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)
  );
}

function formatDateTime(date: Date): string {
  return new Intl.DateTimeFormat("en-IE", {
    dateStyle: "full",
    timeStyle: "short",
    timeZone: CONSULTATION_TIME_ZONE,
  }).format(date);
}

function getSiteUrl(): string {
  return (
    process.env.NEXT_PUBLIC_SITE_URL?.replace(/\/+$/, "") ||

  );
}

export async function POST(request: Request) {
  let createdEventId: string | null = null;
  let calendarIdForCleanup: string | null = null;
  let bookingPersisted = false;

  try {
    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        { error: "Request body must be valid JSON." },
        { status: 400 },
      );
    }

    if (!isRecord(body)) {
      return NextResponse.json(
        { error: "Invalid request body." },
        { status: 400 },
      );
    }

    const assessmentId =
      typeof body.assessmentId === "string"
        ? body.assessmentId.trim()
        : "";

    const submittedName =
      typeof body.attendeeName === "string"
        ? body.attendeeName.trim()
        : "";

    const submittedEmail =
      typeof body.attendeeEmail === "string"
        ? body.attendeeEmail.trim().toLowerCase()
        : "";

    const startAtRaw =
      typeof body.startAt === "string"
        ? body.startAt
        : "";

    const startAt = new Date(startAtRaw);

    if (
      !assessmentId ||
      !Number.isFinite(startAt.getTime()) ||
      !isValidConsultationSlot(startAt)
    ) {
      return NextResponse.json(
        {
          error:
            "Provide a valid assessment and choose a valid future consultation slot.",
        },
        { status: 400 },
      );
    }

    if (
      (submittedName && (
        submittedName.length < 2 ||
        submittedName.length > 120
      )) ||
      (submittedEmail && !isValidEmail(submittedEmail))
    ) {
      return NextResponse.json(
        { error: "Please provide a valid name and email address." },
        { status: 400 },
      );
    }

    const assessment = await prisma.assessment.findUnique({
      where: { id: assessmentId },
      select: {
        id: true,
        status: true,
        qualificationStatus: true,
        businessName: true,
        assessmentSummary: true,
        contactName: true,
        contactEmail: true,
        funnelState: true,
      },
    });

    if (
      !assessment ||
      assessment.status !== "COMPLETED" ||
      assessment.qualificationStatus !== "QUALIFIED"
    ) {
      return NextResponse.json(
        { error: "This assessment is not eligible for booking." },
        { status: 403 },
      );
    }

    // Prefer contact details captured during the assessment funnel.
    // If they exist, do not allow a booking request to substitute a
    // different identity or email address.
    const attendeeName =
      assessment.contactName?.trim() || submittedName;

    const attendeeEmail =
      assessment.contactEmail?.trim().toLowerCase() ||
      submittedEmail;

    if (
      !attendeeName ||
      attendeeName.length < 2 ||
      attendeeName.length > 120 ||
      !attendeeEmail ||
      !isValidEmail(attendeeEmail)
    ) {
      return NextResponse.json(
        {
          error:
            "Please complete the assessment contact details before booking.",
        },
        { status: 400 },
      );
    }

    if (
      assessment.contactEmail &&
      submittedEmail &&
      submittedEmail !== assessment.contactEmail.trim().toLowerCase()
    ) {
      return NextResponse.json(
        {
          error:
            "Please book using the email address associated with your assessment.",
        },
        { status: 400 },
      );
    }

    const endAt = new Date(
      startAt.getTime() + CONSULTATION_MINUTES * 60_000,
    );

    // Do not permit a second active booking for the same assessment.
    const assessmentBooking = await prisma.booking.findFirst({
      where: {
        assessmentId,
        status: "BOOKED",
      },
      select: {
        id: true,
        startAt: true,
      },
    });

    if (assessmentBooking) {
      return NextResponse.json(
        {
          error:
            "This assessment already has a confirmed booking. Please contact the 6ixtem team if you need to change it.",
          existingBooking: {
            id: assessmentBooking.id,
            startAt: assessmentBooking.startAt,
          },
        },
        { status: 409 },
      );
    }

    // Check the database for overlapping active bookings.
    const existing = await prisma.booking.findFirst({
      where: {
        status: "BOOKED",
        startAt: { lt: endAt },
        endAt: { gt: startAt },
      },
      select: { id: true },
    });

    if (existing) {
      return NextResponse.json(
        {
          error:
            "That slot has just been booked. Please choose another time.",
        },
        { status: 409 },
      );
    }

    // Recheck the connected Google Calendar before creating the event.
    const busy = await getBusyIntervals(startAt, endAt);

    const overlaps = busy.some(
      (interval) =>
        startAt < interval.end && endAt > interval.start,
    );

    if (overlaps) {
      return NextResponse.json(
        {
          error:
            "That time is no longer available. Please choose another slot.",
        },
        { status: 409 },
      );
    }

    const { calendar, calendarId } = await getCalendarClient();
    calendarIdForCleanup = calendarId;

    const eventResponse = await calendar.events.insert({
      calendarId,
      conferenceDataVersion: 1,
      sendUpdates: "all",
      requestBody: {
        summary: `6ixtem Consultation${
          assessment.businessName
            ? ` — ${assessment.businessName}`
            : ""
        }`,
        description: [
          "6ixtem business operations consultation.",
          `Assessment reference: ${assessment.id}`,
          assessment.assessmentSummary
            ? `Assessment summary: ${assessment.assessmentSummary.slice(0, 1500)}`
            : "",
        ]
          .filter(Boolean)
          .join("\n\n"),
        start: {
          dateTime: startAt.toISOString(),
          timeZone: CONSULTATION_TIME_ZONE,
        },
        end: {
          dateTime: endAt.toISOString(),
          timeZone: CONSULTATION_TIME_ZONE,
        },
        attendees: [
          {
            email: attendeeEmail,
            displayName: attendeeName,
          },
        ],
        conferenceData: {
          createRequest: {
            requestId: crypto.randomUUID(),
            conferenceSolutionKey: {
              type: "hangoutsMeet",
            },
          },
        },
      },
    });

    createdEventId = eventResponse.data.id ?? null;

    if (!createdEventId) {
      throw new Error(
        "Google Calendar did not return an event ID.",
      );
    }

    // Conference creation may be asynchronous. Retrieve the event
    // again to obtain a Meet URL when it is already available.
    let refreshedEvent = eventResponse.data;

    try {
      const refreshed = await calendar.events.get({
        calendarId,
        eventId: createdEventId,
      });

      refreshedEvent = refreshed.data;
    } catch (error) {
      console.warn(
        "Could not refresh Google Calendar event details yet:",
        error,
      );
    }

    const meetUrl =
      refreshedEvent.hangoutLink ??
      refreshedEvent.conferenceData?.entryPoints?.find(
        (entry) => entry.entryPointType === "video",
      )?.uri ??
      null;

    // Persist the booking before attempting email delivery.
    const booking = await prisma.booking.create({
      data: {
        assessmentId,
        attendeeName,
        attendeeEmail,
        startAt,
        endAt,
        timeZone: CONSULTATION_TIME_ZONE,
        status: "BOOKED",
        googleEventId: createdEventId,
        meetUrl,
      },
      select: {
        id: true,
        assessmentId: true,
        attendeeName: true,
        attendeeEmail: true,
        startAt: true,
        endAt: true,
        timeZone: true,
        status: true,
        meetUrl: true,
        googleEventId: true,
      },
    });

    bookingPersisted = true;

    // The booking is now committed. Email errors are caught separately
    // and must never cause deletion of the confirmed calendar event.
    const emailResults = await sendBookingEmails({
      bookingId: booking.id,
      assessmentId,
      attendeeName,
      attendeeEmail,
      businessName: assessment.businessName,
      startAt,
      endAt,
      meetUrl,
    });

    const message = emailResults.attendeeEmailSent
      ? meetUrl
        ? "Your consultation is booked. A confirmation email and calendar invitation have been sent."
        : "Your consultation is booked and a confirmation email has been sent. Google Meet details may take a short time to become available."
      : "Your consultation is booked, but we could not send the confirmation email. Your booking remains confirmed; please keep the booking reference and contact the 6ixtem team if needed.";

    return NextResponse.json(
      {
        booking,
        calendarEventId: createdEventId,
        meetUrl,
        email: {
          attendeeSent: emailResults.attendeeEmailSent,
          adminSent: emailResults.adminEmailSent,
        },
        message,
      },
      { status: 201 },
    );
  } catch (error) {
    console.error("6ixtem booking error:", error);

    // Only remove the Google event if booking persistence did not
    // succeed. Email failures happen after persistence and are handled
    // above, so they do not delete an otherwise valid booking.
    if (
      createdEventId &&
      calendarIdForCleanup &&
      !bookingPersisted
    ) {
      try {
        const { calendar } = await getCalendarClient();

        await calendar.events.delete({
          calendarId: calendarIdForCleanup,
          eventId: createdEventId,
          sendUpdates: "all",
        });
      } catch (cleanupError) {
        console.error(
          "Could not clean up the Google Calendar event:",
          cleanupError,
        );
      }
    }

    return NextResponse.json(
      {
        error:
          "We could not complete the booking. Please refresh availability and try again.",
      },
      { status: 500 },
    );
  }
}

export function escapeHtml(value: string): string {
  return value.replace(/[&<>"']/g, (character) => {
    const entities: Record<string, string> = {
      "&": "&amp;",
      "<": "&lt;",
      ">": "&gt;",
      '"': "&quot;",
      "'": "&#39;",
    };

    return entities[character];
  });
}

export function getAdminNotificationEmail(): string | null {
  const email = ADMIN_EMAIL?.trim();

  return email && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
    ? email
    : null;
}

/**
 * Store an email before attempting delivery.
 * The unique key prevents duplicate queue records.
 */
export async function queue6ixtemEmail(input: {
  key: string;
  type: EmailDeliveryType;
  bookingId?: string | null;
  to: string;
  subject: string;
  html: string;
  text: string;
  scheduledFor?: Date;
}) {
  const to = input.to.trim();

  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(to)) {
    throw new Error("A valid recipient email is required.");
  }

  return prisma.emailDelivery.upsert({
    where: {
      key: input.key,
    },
    create: {
      key: input.key,
      type: input.type,
      bookingId: input.bookingId ?? null,
      recipient: to,
      subject: input.subject,
      html: input.html,
      text: input.text,
      scheduledFor: input.scheduledFor ?? new Date(),
      status: "PENDING",
    },
    update: {},
    select: {
      id: true,
      key: true,
      status: true,
    },
  });
}

/**
 * Send a single email.
 * Provider errors are returned rather than silently treated as success.
 */
export async function send6ixtemEmail(
  input: SendEmailInput,
): Promise<EmailResult> {
  if (!resend) {
    return {
      success: false,
      error: "Email provider is not configured.",
    };
  }

  if (
    !input.to ||
    !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(input.to.trim())
  ) {
    return {
      success: false,
      error: "A valid recipient email is required.",
    };
  }

  try {
    const result = await resend.emails.send({
      from: FROM_EMAIL,
      to: input.to.trim(),
      subject: input.subject,
      html: input.html,
      text: input.text,
    });

    if (result.error) {
      console.error("Resend email error:", result.error);

      return {
        success: false,
        error: result.error.message || "Resend rejected the email.",
      };
    }

    return {
      success: true,
      id: result.data?.id ?? null,
    };
  } catch (error) {
    console.error(
      "6ixtem email exception:",
      error instanceof Error ? error.message : String(error),
    );

    return {
      success: false,
      error: "The email could not be sent.",
    };
  }
}
