function escapeHtml(value: string): string {
  return value.replace(/[&<>"']/g, (character) => {
    const entities: Record<string, string> = {
      "&": "&amp;",
      "<": "&lt;",
      ">": "&gt;",
      '"': "&quot;",
      "'": "&#39;",
    };

    return entities[character];
  });
}

export async function send6ixtemEmail(
  input: SendEmailInput,
): Promise<EmailResult> {
  if (!resend) {
    console.error("6ixtem email not sent: RESEND_API_KEY is missing.");
    return {
      success: false,
      error: "Email provider is not configured.",
    };
  }

  if (!input.to || !input.to.includes("@")) {
    return {
      success: false,
      error: "A valid recipient email is required.",
    };
  }

  try {
    const result = await resend.emails.send({
      from: FROM_EMAIL,
      to: input.to,
      subject: input.subject,
      html: input.html,
      text: input.text,
    });

    if (result.error) {
      console.error("Resend email error:", result.error);
      return {
        success: false,
        error: result.error.message || "Resend rejected the email.",
      };
    }

    return {
      success: true,
      id: result.data?.id ?? null,
    };
  } catch (error) {
    console.error(
      "6ixtem email exception:",
      error instanceof Error ? error.message : String(error),
    );

    return {
      success: false,
      error: "The email could not be sent.",
    };
  }
}

import "server-only";

import crypto from "node:crypto";
import { google, type calendar_v3 } from "googleapis";
import { prisma } from "@/lib/prisma";

export const CONSULTATION_TIME_ZONE = "Africa/Lagos";
export const CONSULTATION_MINUTES = 30;

const WORKDAY_START_MINUTES = 9 * 60;
const WORKDAY_END_MINUTES = 17 * 60;

function requiredEnv(name: string): string {
  const value = process.env[name];

  if (!value) {
    throw new Error(
      `Missing required environment variable: ${name}`,
    );
  }

  return value;
}

function getOAuthClient() {
  return new google.auth.OAuth2(
    requiredEnv("GOOGLE_CLIENT_ID"),
    requiredEnv("GOOGLE_CLIENT_SECRET"),
    requiredEnv("GOOGLE_REDIRECT_URI"),
  );
}

function encryptionKey(): Buffer {
  const key = requiredEnv(
    "GOOGLE_TOKEN_ENCRYPTION_KEY",
  );

  if (!/^[a-fA-F0-9]{64}$/.test(key)) {
    throw new Error(
      "GOOGLE_TOKEN_ENCRYPTION_KEY must contain 64 hexadecimal characters.",
    );
  }

  return Buffer.from(key, "hex");
}

function encryptToken(token: string): string {
  const iv = crypto.randomBytes(12);

  const cipher = crypto.createCipheriv(
    "aes-256-gcm",
    encryptionKey(),
    iv,
  );

  const ciphertext = Buffer.concat([
    cipher.update(token, "utf8"),
    cipher.final(),
  ]);

  const tag = cipher.getAuthTag();

  return [
    iv.toString("base64url"),
    tag.toString("base64url"),
    ciphertext.toString("base64url"),
  ].join(".");
}

function decryptToken(payload: string): string {
  const [ivPart, tagPart, ciphertextPart] =
    payload.split(".");

  if (!ivPart || !tagPart || !ciphertextPart) {
    throw new Error(
      "Stored Google refresh token has an invalid format.",
    );
  }

  const decipher = crypto.createDecipheriv(
    "aes-256-gcm",
    encryptionKey(),
    Buffer.from(ivPart, "base64url"),
  );

  decipher.setAuthTag(
    Buffer.from(tagPart, "base64url"),
  );

  return Buffer.concat([
    decipher.update(
      Buffer.from(ciphertextPart, "base64url"),
    ),
    decipher.final(),
  ]).toString("utf8");
}

export function createOAuthState(): string {
  const timestamp = Date.now().toString();

  const nonce = crypto
    .randomBytes(24)
    .toString("base64url");

  const payload = `${timestamp}.${nonce}`;

  const signature = crypto
    .createHmac(
      "sha256",
      requiredEnv("GOOGLE_CALENDAR_CONNECT_SECRET"),
    )
    .update(payload)
    .digest("base64url");

  return `${payload}.${signature}`;
}

export function verifyOAuthState(
  state: string | null,
): boolean {
  if (!state) return false;

  const parts = state.split(".");

  if (parts.length !== 3) return false;

  const [timestamp, nonce, signature] = parts;

  if (
    !/^\d+$/.test(timestamp) ||
    !nonce ||
    !signature
  ) {
    return false;
  }

  const age = Date.now() - Number(timestamp);

  if (age < 0 || age > 10 * 60 * 1000) {
    return false;
  }

  const expected = crypto
    .createHmac(
      "sha256",
      requiredEnv("GOOGLE_CALENDAR_CONNECT_SECRET"),
    )
    .update(`${timestamp}.${nonce}`)
    .digest();

  const actual = Buffer.from(
    signature,
    "base64url",
  );

  return (
    actual.length === expected.length &&
    crypto.timingSafeEqual(actual, expected)
  );
}

export async function getCalendarClient(): Promise<{

function datePartsInZone(date: Date) {
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone: CONSULTATION_TIME_ZONE,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    hourCycle: "h23",
    weekday: "short",
  }).formatToParts(date);

  const map = Object.fromEntries(
    parts.map((part) => [
      part.type,
      part.value,
    ]),
  );

  const weekdays: Record<string, number> = {
    Mon: 1,
    Tue: 2,
    Wed: 3,
    Thu: 4,
    Fri: 5,
    Sat: 6,
    Sun: 7,
  };

  return {
    year: Number(map.year),
    month: Number(map.month),
    day: Number(map.day),
    hour: Number(map.hour),
    minute: Number(map.minute),
    weekday: weekdays[map.weekday] ?? 0,
  };
}

/**
 * Convert a Lagos local date and time to a UTC instant.
 * Nigeria uses UTC+1 and does not observe daylight saving time.
 */
function localDateToUtc(
  dateString: string,
  hour: number,
  minute = 0,
): Date {
  const match =
    /^(\d{4})-(\d{2})-(\d{2})$/.exec(dateString);

  if (!match) {
    throw new Error(
      "Date must use YYYY-MM-DD.",
    );
  }

  const [, year, month, day] = match;

  return new Date(
    Date.UTC(
      Number(year),
      Number(month) - 1,
      Number(day),
      hour - 1,
      minute,
    ),
  );
}

export function localDateRange(dateString: string) {
  return {
    start: localDateToUtc(dateString, 0),
    end: localDateToUtc(
      addLocalDays(dateString, 1),
      0,
    ),
  };
}

export function addLocalDays(
  dateString: string,
  days: number,
): string {
  const match =
    /^(\d{4})-(\d{2})-(\d{2})$/.exec(dateString);

  if (!match) {
    throw new Error(
      "Date must use YYYY-MM-DD.",
    );
  }

  const date = new Date(
    Date.UTC(
      Number(match[1]),
      Number(match[2]) - 1,
      Number(match[3]) + days,
    ),
  );

  return [
    date.getUTCFullYear(),
    String(date.getUTCMonth() + 1).padStart(2, "0"),
    String(date.getUTCDate()).padStart(2, "0"),
  ].join("-");
}

export function isValidLocalDate(
  dateString: string,
): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(dateString)) {
    return false;
  }

  const date = new Date(
    `${dateString}T00:00:00Z`,
  );

  return (
    Number.isFinite(date.getTime()) &&
    date.toISOString().slice(0, 10) === dateString
  );
}

export function isValidConsultationSlot(
  start: Date,
): boolean {
  if (!Number.isFinite(start.getTime())) {
    return false;
  }

  const parts = datePartsInZone(start);

  const minutes =
    parts.hour * 60 + parts.minute;

  return (
    parts.weekday >= 1 &&
    parts.weekday <= 5 &&
    parts.minute === 0 &&
    minutes >= WORKDAY_START_MINUTES &&
    minutes + CONSULTATION_MINUTES <=
      WORKDAY_END_MINUTES &&
    minutes % CONSULTATION_MINUTES === 0 &&
    start.getTime() > Date.now()
  );
}

export function createSlotsForDate(
  dateString: string,
  busy: Array<{
    start: Date;
    end: Date;
  }>,
) {
  const range = localDateRange(dateString);

  // Noon in Lagos reliably identifies the weekday
  // for the requested local calendar date.
  const weekdayParts = datePartsInZone(
    new Date(
      range.start.getTime() +
        12 * 60 * 60 * 1000,
    ),
  );

  if (
    weekdayParts.weekday < 1 ||
    weekdayParts.weekday > 5
  ) {
    return [];
  }

  const slots: Array<{
    startAt: string;
    endAt: string;
    label: string;
  }> = [];

  for (
    let minute = WORKDAY_START_MINUTES;
    minute + CONSULTATION_MINUTES <=
    WORKDAY_END_MINUTES;
    minute += CONSULTATION_MINUTES
  ) {
    const start = localDateToUtc(
      dateString,
      Math.floor(minute / 60),
      minute % 60,
    );

    const end = new Date(
      start.getTime() +
        CONSULTATION_MINUTES * 60_000,
    );

    if (start.getTime() <= Date.now()) {
      continue;
    }

    const overlaps = busy.some(
      (interval) =>
        start.getTime() < interval.end.getTime() &&
        end.getTime() > interval.start.getTime(),
    );

    if (overlaps) continue;

    slots.push({
      startAt: start.toISOString(),
      endAt: end.toISOString(),

      label: new Intl.DateTimeFormat("en-NG", {
        timeZone: CONSULTATION_TIME_ZONE,
        hour: "numeric",
        minute: "2-digit",
      }).format(start),
    });
  }

  return slots;
}

export async function getBusyIntervals(
  start: Date,
  end: Date,
) {
  const { calendar, calendarId } =
    await getCalendarClient();

  const [freeBusy, bookings] = await Promise.all([
    calendar.freebusy.query({
      requestBody: {
        timeMin: start.toISOString(),
        timeMax: end.toISOString(),
        timeZone: CONSULTATION_TIME_ZONE,
        items: [
          {
            id: calendarId,
          },
        ],
      },
    }),

    prisma.booking.findMany({
      where: {
        status: "BOOKED",
        startAt: {
          lt: end,
        },
        endAt: {
          gt: start,
        },
      },

      select: {
        startAt: true,
        endAt: true,
      },
    }),
  ]);

  const googleBusy =
    freeBusy.data.calendars?.[calendarId]?.busy ?? [];

  return [
    ...googleBusy
      .filter(
        (item) => item.start && item.end,
      )
      .map((item) => ({
        start: new Date(item.start!),
        end: new Date(item.end!),
      })),

    ...bookings.map((booking) => ({
      start: booking.startAt,
      end: booking.endAt,
    })),
  ];
}

        submittedName.length < 2 ||
        submittedName.length > 120
      )) ||
      (submittedEmail && !isValidEmail(submittedEmail))

        { status: 400 },

        contactName: true,
        contactEmail: true,
        funnelState: true,

      assessment.status !== "COMPLETED" ||
      assessment.qualificationStatus !== "QUALIFIED"

        { error: "This assessment is not eligible for booking." },
        { status: 403 },
      );

    // If they exist, do not allow a booking request to substitute a
    // different identity or email address.

      assessment.contactName?.trim() || submittedName;

    const attendeeEmail =
      assessment.contactEmail?.trim().toLowerCase() ||

      !attendeeName ||
      attendeeName.length < 2 ||
      attendeeName.length > 120 ||

        attendeeName,

      attendeeName,
      attendeeEmail,

        : "Your consultation is booked and a confirmation email has been sent. Google Meet details may take a short time to become available."
      : "Your consultation is booked, but we could not send the confirmation email. Your booking remains confirmed; please keep the booking reference and contact the 6ixtem team if needed.";

    return NextResponse.json(
      {
        booking,
        calendarEventId: createdEventId,
        meetUrl,
        email: {
          attendeeSent: emailResults.attendeeEmailSent,

        message,
      },
      { status: 201 },
    );
  } catch (error) {
    console.error("6ixtem booking error:", error);

    // Only remove the Google event if booking persistence did not
    // succeed. Email failures happen after persistence and are handled
    // above, so they do not delete an otherwise valid booking.
    if (
      createdEventId &&
      calendarIdForCleanup &&
      !bookingPersisted

        });

          "Could not clean up the Google Calendar event:",
          cleanupError,
        );
      }
    }

    return NextResponse.json(
      {
        error:
          "We could not complete the booking. Please refresh availability and try again.",
      },
      { status: 500 },
    );
  }
}

import {
  addLocalDays,
  createSlotsForDate,
  getBusyIntervals,
  isValidLocalDate,
  localDateRange,
} from "@/lib/6ixtem-google-calendar";

export async function GET(request: Request) {
  try {
    const url = new URL(request.url);

    const assessmentId =
      url.searchParams.get("assessmentId") ?? "";

    const from =
      url.searchParams.get("from") ?? "";

    const days = Number(
      url.searchParams.get("days") ?? "14",
    );

    if (
      !assessmentId ||
      !isValidLocalDate(from) ||
      !Number.isInteger(days) ||
      days < 1 ||
      days > 21
    ) {
      return NextResponse.json(
        {
          error:
            "Provide a valid assessmentId, a from date in YYYY-MM-DD format, and days between 1 and 21.",
        },
        { status: 400 },
      );
    }

    const assessment =
      await prisma.assessment.findUnique({
        where: {
          id: assessmentId,
        },

        select: {
          status: true,
          qualificationStatus: true,
        },
      });

    if (
      !assessment ||
      assessment.status !== "COMPLETED" ||
      assessment.qualificationStatus !== "QUALIFIED"
    ) {
      return NextResponse.json(
        {
          error:
            "Only completed, qualified assessments can book a consultation.",
        },
        { status: 403 },
      );
    }

    const rangeStart =
      localDateRange(from).start;

    const rangeEnd =
      localDateRange(
        addLocalDays(from, days),
      ).start;

    const busy = await getBusyIntervals(
      rangeStart,
      rangeEnd,
    );

    const dates: Array<{
      date: string;
      slots: ReturnType<
        typeof createSlotsForDate
      >;
    }> = [];

    for (let i = 0; i < days; i++) {
      const date = addLocalDays(from, i);

      dates.push({
        date,
        slots: createSlotsForDate(
          date,
          busy,
        ),
      });
    }

    return NextResponse.json({
      timeZone: "Africa/Lagos",
      durationMinutes: 30,
      dates,
    });
  } catch (error) {
    console.error(
      "6ixtem availability error:",
      error,
    );

    return NextResponse.json(
      {
        error:
          "Could not retrieve calendar availability.",
      },
      { status: 503 },
    );
  }
}

export async function POST(request: Request) {

  }

    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "Invalid JSON request body." },
      { status: 400 },
    );
  }

  if (
    typeof body !== "object" ||
    body === null ||
    !("bookingId" in body) ||
    typeof body.bookingId !== "string" ||
    !body.bookingId.trim()
  ) {
    return NextResponse.json(
      { error: "A valid bookingId is required." },
      { status: 400 },
    );
  }

  const bookingId = body.bookingId.trim();

  try {

        googleEventId: true,

      },
    });

      booking.cancelledAt
    ) {
      return NextResponse.json({
        success: true,
        message: "The booking is already cancelled.",
      });
    }

    if (booking.status !== BookingStatus.BOOKED) {
      return NextResponse.json(
        {
          error: `Cannot cancel a booking with status ${booking.status}.`,
        },
        { status: 409 },
      );
    }

    // Remove the calendar event first. If Google fails unexpectedly,
    // leave the database booking unchanged and retry the operation.
    if (booking.googleEventId) {
      const { calendar, calendarId } = await getCalendarClient();

      try {
        await calendar.events.delete({
          calendarId,
          eventId: booking.googleEventId,
          sendUpdates: "all",
        });
      } catch (error) {
        const status = (
          error as { code?: number; response?: { status?: number } }
        )?.response?.status ??
          (error as { code?: number })?.code;

        // A missing Google event is already effectively deleted.
        if (status !== 404 && status !== 410) {
          throw error;
        }
      }
    }

    const updated = await prisma.booking.updateMany({
      where: {
        id: booking.id,
        status: BookingStatus.BOOKED,
        cancelledAt: null,
      },
      data: {
        status: BookingStatus.CANCELLED,
        cancelledAt: new Date(),
      },
    });

    if (updated.count !== 1) {
      return NextResponse.json(
        {
          error:
            "The booking changed while cancellation was in progress. Refresh and check its current status.",
        },
        { status: 409 },
      );
    }

    return NextResponse.json({
      success: true,
      bookingId: booking.id,
      status: BookingStatus.CANCELLED,
      message: "The booking has been cancelled.",
    });
  } catch (error) {
    console.error("6ixtem booking cancellation failed:", error);

    return NextResponse.json(
      {
        error:
          "Cancellation could not be completed. Check the booking and calendar before retrying.",
      },
      { status: 500 },
    );
  }
}

      success: true,
      previousBookingId: oldBooking.id,
      previousStatus: BookingStatus.RESCHEDULED,
      booking: replacement,
      message:
        "The consultation has been rescheduled. The replacement booking is saved in the database.",
    });
  } catch (error) {
    if (newGoogleEventId && !databaseCommitted) {
      try {
        const { calendar, calendarId } = await getCalendarClient();

        await calendar.events.delete({
          calendarId,
          eventId: newGoogleEventId,
          sendUpdates: "all",
        });
      } catch (cleanupError) {
        console.error(
          "Could not remove the uncommitted replacement event:",
          cleanupError,
        );
      }
    }

    if (
      error instanceof Error &&
      error.message === "BOOKING_STATE_CHANGED"
    ) {
      return NextResponse.json(
        {
          error:
            "The original booking changed during rescheduling. Refresh and try again.",
        },
        { status: 409 },
      );
    }

    if (
      error instanceof Error &&
      error.message === "SLOT_BECAME_UNAVAILABLE"
    ) {
      return NextResponse.json(
        {
          error:
            "That slot was booked by someone else. Choose another time.",
        },
        { status: 409 },
      );
    }

    console.error("6ixtem rescheduling failed:", error);

    return NextResponse.json(
      {
        error:
          "The consultation could not be rescheduled. Check the booking and calendar before retrying.",
      },
      { status: 500 },
    );
  }
}

import crypto from "node:crypto";
import { NextResponse } from "next/server";

import {
  buildOAuthClient,
  CALENDAR_SCOPE,
  createOAuthState,
} from "@/lib/6ixtem-google-calendar";

export async function GET(request: Request) {
  const url = new URL(request.url);

  const suppliedSecret =
    url.searchParams.get("secret");

  const expectedSecret =
    process.env.GOOGLE_CALENDAR_CONNECT_SECRET;

  if (
    !expectedSecret ||
    !suppliedSecret ||
    Buffer.byteLength(suppliedSecret) !==
      Buffer.byteLength(expectedSecret)
  ) {
    return NextResponse.json(
      { error: "Not authorized." },
      { status: 401 },
    );
  }

  if (
    !crypto.timingSafeEqual(
      Buffer.from(suppliedSecret),
      Buffer.from(expectedSecret),
    )
  ) {
    return NextResponse.json(
      { error: "Not authorized." },
      { status: 401 },
    );
  }

  try {
    const oauth = buildOAuthClient();
    const state = createOAuthState();

    const authUrl = oauth.generateAuthUrl({
      access_type: "offline",
      prompt: "consent",

      scope: [
        CALENDAR_SCOPE,
        "https://www.googleapis.com/auth/userinfo.email",
      ],

      state,
    });

    return NextResponse.redirect(authUrl);
  } catch (error) {
    console.error(
      "Google Calendar connection error:",
      error,
    );

    return NextResponse.json(
      { error: "Could not start Google authorization." },
      { status: 500 },
    );
  }
}

import { NextResponse } from "next/server";
import { google } from "googleapis";

import {
  buildOAuthClient,
  saveGoogleTokens,
  verifyOAuthState,
} from "@/lib/6ixtem-google-calendar";

export async function GET(request: Request) {
  const url = new URL(request.url);

  const code = url.searchParams.get("code");
  const state = url.searchParams.get("state");
  const oauthError = url.searchParams.get("error");

  if (oauthError) {
    return NextResponse.json(
      {
        error:
          "Google Calendar authorization was declined.",
      },
      { status: 400 },
    );
  }

  if (
    !code ||
    !verifyOAuthState(state)
  ) {
    return NextResponse.json(
      {
        error:
          "Invalid or expired Google OAuth state.",
      },
      { status: 400 },
    );
  }

  try {
    const oauth = buildOAuthClient();

    const { tokens } = await oauth.getToken(code);

    if (!tokens.refresh_token) {
      return NextResponse.json(
        {
          error:
            "Google did not return a refresh token. Revoke the application's access in your Google Account, then connect again.",
        },
        { status: 400 },
      );
    }

    oauth.setCredentials(tokens);

    const oauth2 = google.oauth2({
      version: "v2",
      auth: oauth,
    });

    const profile =
      await oauth2.userinfo.get();

    await saveGoogleTokens(
      tokens.refresh_token,
      profile.data.email,
    );

    const baseUrl =
      process.env.NEXT_PUBLIC_SITE_URL ||
      url.origin;

    return NextResponse.redirect(
      new URL(
        "/6ixtem/assessment/booking?calendar=connected",
        baseUrl,
      ),
    );
  } catch (error) {
    console.error(
      "Google Calendar OAuth callback failed:",
      error,
    );

    return NextResponse.json(
      {
        error:
          "Could not connect Google Calendar.",
      },
      { status: 500 },
    );
  }
}
